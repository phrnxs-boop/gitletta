// ─── Staff Authentication System ──────────────────────────────────────────
// Completely separate from the Owner login. Staff use a 4-digit PIN.
// PINs are hashed (never stored in plaintext). Rate limiting + lockout.
model Staff {
  id              String   @id @default(cuid())
  tenantId        String
  name            String
  employeeId      String   // username / employee ID for login
  roleId          String?
  pinHash        String   // hashed PIN (never plaintext)
  pinSalt        String   // per-staff salt
  active          Boolean  @default(true)
  avatar          String?
  // rate limiting
  failedAttempts  Int      @default(0)
  lockedUntil     DateTime?
  lastLogin       DateTime?
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  tenant      Tenant          @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  role        Role?           @relation(fields: [roleId], references: [id])
  sessions    StaffSession[]
  activities  StaffActivity[]

  @@unique([tenantId, employeeId])
  @@index([tenantId])
}

model StaffSession {

model Table {
  id        String   @id @default(cuid())

// A table session is created when a customer scans a table's QR code.
// It is the BACKEND source of truth for whether a customer may view the menu / place orders.
// A session is ACTIVE until a staff member ends it (or it's auto-ended). Once ENDED, the
// menu/order APIs reject all requests referencing that session.
model TableSession {
  id          String   @id @default(cuid())
  tenantId    String
  tableId     String
  token       String   @unique @default(cuid()) // handed to the client; must be re-validated on every request
  status      String   @default("ACTIVE") // ACTIVE | ENDED
  deviceFp    String?  // best-effort device fingerprint (informational only; backend is source of truth)
  createdAt   DateTime @default(now())
  endedAt     DateTime?

  tenant      Tenant    @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  table       Table     @relation(fields: [tableId], references: [id], onDelete: Cascade)
  orders      Order[]

  @@index([tenantId, tableId, status])
  @@index([token])
}

model Order {
  id            String   @id @default(cuid())
  tenantId      String
  orderNumber   Int
  tableId       String?
  orderType     String   @default("DINE_IN") // DINE_IN, TAKEAWAY, DELIVERY
  status        String   @default("PENDING") // PENDING, PREPARING, READY, SERVED, COMPLETED, CANCELLED
  itemsTotal    Float    @default(0)
  discount      Float    @default(0)
  tax           Float    @default(0)
  serviceCharge Float    @default(0)
  total         Float    @default(0)
  promoCodeId   String?
  promoCode     String?
  customerName  String?
  customerPhone String?
  notes         String?
  paymentMethod String?  // CASH, CARD, WALLET
  paymentStatus String   @default("UNPAID") // UNPAID, PAID, REFUNDED
  servedById    String?
  tableSessionId String? // links to the active TableSession that created this order (historical; NOT deleted when session ends)
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt
  completedAt   DateTime?

  tenant      Tenant      @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  table       Table?      @relation(fields: [tableId], references: [id])
  items       OrderItem[]
  servedBy    User?       @relation(fields: [servedById], references: [id])
  session     TableSession? @relation(fields: [tableSessionId], references: [id])

  @@index([tenantId, status])
  @@index([tenantId, createdAt])
}

// ─── Google Reviews integration ──────────────────────────────────────────
// A connection represents an authorized Google Business Profile for a tenant.
// It is fully independent from customer table sessions — ending a table
// session NEVER touches this data.
model GoogleConnection {
  id                 String   @id @default(cuid())
  tenantId           String
  // OAuth credentials (encrypted-at-rest in production).
  accessToken        String
  refreshToken       String?
  tokenExpiresAt     DateTime?
  // The Google Business Profile location this connection maps to
  accountName        String?
  locationName       String?
  locationId         String?
  placeId            String?
  // feature config
  enabled            Boolean  @default(true)
  autoSync           Boolean  @default(true)
  syncIntervalMins   Int      @default(60)
  showPublic         Boolean  @default(true)   // show reviews on public menu
  minRatingFilter    Int      @default(0)      // 0 = all, 1-5 = only show ≥ N stars
  lastSyncedAt       DateTime?
  lastSyncError      String?
  createdAt          DateTime @default(now())
  updatedAt          DateTime @updatedAt

  tenant      Tenant         @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  reviews     GoogleReview[]

  @@index([tenantId])
}

// Cached Google reviews for a tenant. Synced from the Google Business Profile API.
// Historical — reviews are never deleted when a table session ends.
model GoogleReview {
  id              String   @id @default(cuid())
  tenantId        String
  connectionId    String
  googleReviewId  String   // Google's review ID (unique per Google review)
  authorName      String?
  authorPhotoUrl  String?
  rating          Int      // 1-5
  comment         String?
  commentTranslated String?
  createTime      DateTime?
  updateTime      DateTime?
  reply           String?  // owner reply (if any)
  replyTime       DateTime?
  // local state
  flagged         Boolean  @default(false)
  syncedAt        DateTime @default(now())

  tenant       Tenant            @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  connection   GoogleConnection  @relation(fields: [connectionId], references: [id], onDelete: Cascade)

  @@unique([tenantId, googleReviewId])
  @@index([tenantId, rating])
}

model CustomerReview {
  id         String   @id @default(cuid())

  rating     Int      @default(5) // 1-5

  table      Table?   @relation(fields: [tableId], references: [id], onDelete: SetNull)
